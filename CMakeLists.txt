cmake_minimum_required(VERSION 3.20)
project(ProjectName LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)

FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.5.1)
FetchContent_MakeAvailable(glfw)

file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/*.cpp")
file(GLOB LIBRARIES CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/third_party/*")
foreach(LIBRARY ${LIBRARIES})
    if (IS_DIRECTORY ${LIBRARY})
        file(GLOB_RECURSE CURRENT CONFIGURE_DEPENDS "${LIBRARY}/src/*.c" "${LIBRARY}/src/*.cpp")
        list(APPEND SOURCES ${CURRENT})
    endif()
endforeach()
add_executable(${PROJECT_NAME} ${SOURCES})

target_include_directories(${PROJECT_NAME} PRIVATE "${PROJECT_SOURCE_DIR}/include")
target_compile_definitions(${PROJECT_NAME} PRIVATE PROJECT_ROOT="${PROJECT_SOURCE_DIR}")
target_link_libraries(${PROJECT_NAME} PRIVATE glfw)
foreach(LIBRARY ${LIBRARIES})
    if (IS_DIRECTORY ${LIBRARY})
        target_include_directories(${PROJECT_NAME} PRIVATE "${LIBRARY}/include")
    endif()
endforeach()
